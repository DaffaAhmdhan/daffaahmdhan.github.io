WITH breach_counts AS (
    SELECT root_cause, COUNT(*) AS breach_count FROM fact_tickets WHERE sla_met = 0 GROUP BY root_cause
), total_breaches AS (SELECT SUM(breach_count) AS total FROM breach_counts)
SELECT b.root_cause, ROUND(100.0 * SUM(b.breach_count) OVER (ORDER BY b.breach_count DESC) / t.total, 2) AS cum_pct
FROM breach_counts b CROSS JOIN total_breaches t;
