Pareto 80/20 SQL Query (sql/schema_and_kpis.sql)

WITH breach_counts AS (
    SELECT root_cause, COUNT(*) AS breach_count
    FROM fact_tickets WHERE sla_met = 0 GROUP BY root_cause
),
total_breaches AS (SELECT SUM(breach_count) AS total FROM breach_counts)
SELECT b.root_cause, b.breach_count,
  ROUND(100.0 * SUM(b.breach_count) OVER (ORDER BY b.breach_count DESC) / t.total, 2) AS cumulative_pct
FROM breach_counts b CROSS JOIN total_breaches t ORDER BY b.breach_count DESC;

RFM Customer Aggregation Query (sql/cohort_and_rfm.sql)

SELECT customer_id,
  CAST(julianday('2026-11-01') - julianday(MAX(order_date)) AS INTEGER) AS recency_days,

  SUM(order_value_idr) AS monetary_idr
